{{/* Renders a list of normalized owners (see normalize-owners.html) as GitHub profile/team links. Each owner renders as "{DisplayName} ({Handle})" when a display name is available, otherwise just "{Handle}" - in both cases linking to the GitHub user/team page. input: slice of { Handle, DisplayName, IsTeam } */}} {{- $owners := . -}} {{- if $owners -}} {{- range $index, $owner := $owners -}} {{- if gt $index 0 }}
{{ end -}} {{- $url := printf "https://github.com/%s" $owner.Handle -}} {{- if $owner.IsTeam -}} {{- $teamSlug := strings.TrimPrefix "@Azure/" $owner.Handle -}} {{- $url = printf "https://github.com/orgs/Azure/teams/%s" $teamSlug -}} {{- end -}} {{- $label := $owner.Handle -}} {{- with $owner.DisplayName -}} {{- $label = printf "%s (%s)" . $owner.Handle -}} {{- end -}} {{ $label }} {{- end -}} {{- end -}}
